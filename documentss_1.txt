  # Initialize Git Repository
git init

# Create a file (app.txt)
echo "Hello Git" > app.txt

# Check repository status
git status

# Stage file
git add app.txt

# Commit changes
git commit -m "Initial commit with app.txt"

# Modify file
echo "Updated content in app.txt" >> app.txt

# View differences
git diff

# Stage updated file
git add app.txt

# Commit updated changes
git commit -m "Updated app.txt"

# Create dev branch
git branch dev

# Switch to dev branch
git checkout dev

# Make changes in dev branch
echo "Changes made in dev branch" >> app.txt

# Check differences
git diff

# Stage changes
git add app.txt

# Commit changes in dev
git commit -m "Changes added in dev branch"

# Switch back to main branch
git checkout main

# Compare branches
git diff main dev

# Show all branches
git branch

# View commit history
git log --oneline --graph --all

git merge dev

# Git will show MERGE CONFLICT.

# Open app.txt manually.

# >>>>>>> dev

git commit -m "Resolved merge conflict"

# ---------------- INITIAL SETUP ----------------

# Initialize repository
git init

# Create files manually:
# app.txt
# data.txt

# Add some content inside files

# Check status
git status

# Stage files
git add .

# Commit files
git commit -m "Initial commit"


# ---------------- UNDO UNSTAGED CHANGES ----------------

# Open app.txt manually
# Add or modify some content

# Check status
git status

# Undo unstaged changes
git checkout -- app.txt

# File returns to last committed state


# ---------------- UNSTAGE FILES ----------------

# Modify app.txt again manually

# Stage file
git add app.txt

# Check status
git status

# Unstage file
git reset app.txt

# File remains modified locally
# But removed from staging area


# ---------------- AMEND LAST COMMIT ----------------

# Modify file manually again

# Stage updated file
git add app.txt

# Modify previous commit
git commit --amend -m "Updated initial commit"


# ---------------- GIT RESET SOFT ----------------

# View commits
git log --oneline

# Move HEAD back by 1 commit
# Keep changes staged
git reset --soft HEAD~1


# ---------------- GIT RESET MIXED ----------------

# Default reset mode
# Keep changes in working directory
# Remove from staging area
git reset --mixed HEAD~1

# OR simply:
# git reset HEAD~1


# ---------------- GIT RESET HARD ----------------

# WARNING:
# Deletes commits and local changes permanently

git reset --hard HEAD~1


# ---------------- GIT REVERT ----------------

# Safely undo a commit by creating new reverse commit

# View commits
git log --oneline

# Revert specific commit
git revert COMMIT_ID

# Example:
# git revert a1b2c3d


# ---------------- GIT REFLOG ----------------

# View all HEAD history
git reflog

# Recover lost commit
git reset --hard REFLOG_ID

# Example:
# git reset --hard abc1234


# ---------------- CHECK FINAL STATUS ----------------

git status

# View commit history
git log --oneline --graph --all

# Repository stored in your own system/laptop

# REMOTE REPOSITORY:
# Repository stored online (GitHub)


# =========================================================
# CREATE LOCAL REPOSITORY
# =========================================================

# Create project folder manually
# Example:
# GitProject

# Open terminal inside folder

# Initialize git
git init

# Create app.txt manually
# Add some content

# Check status
git status

# Stage file
git add app.txt

# Commit file
git commit -m "Initial commit"


# =========================================================
# CONFIGURE REMOTE REPOSITORY
# =========================================================

# Create repository manually on GitHub

# Connect local repo with remote repo
git remote add origin https://github.com/USERNAME/repository.git

# Verify remote
git remote -v


# =========================================================
# PUSH CHANGES TO REMOTE
# =========================================================

# Push local commits to GitHub
git push -u origin main


# =========================================================
# CLONE REPOSITORY
# =========================================================

# Clone repository into another directory
git clone https://github.com/USERNAME/repository.git

# Move into cloned repository
cd repository


# =========================================================
# FETCH UPDATES
# =========================================================

# Download remote updates only
git fetch

# View fetched commits
git log --oneline --all --graph


# =========================================================
# PULL CHANGES
# =========================================================

# Download + merge remote changes
git pull origin main


# =========================================================
# MAKE LOCAL CHANGES
# =========================================================

# Open app.txt manually
# Add new content

# Stage changes
git add app.txt

# Commit changes
git commit -m "Updated app file"

# Push changes to GitHub
git push origin main


# =========================================================
# IDENTIFY COMMIT TO ROLLBACK
# =========================================================

# View commit history
git log --oneline

# Example output:
# a1b2c3d Updated app
# x7y8z9 Initial commit


# =========================================================
# RESET LOCAL BRANCH TO PREVIOUS COMMIT
# =========================================================

# Soft reset:
# Removes commit but keeps changes staged
git reset --soft HEAD~1

# Mixed reset:
# Removes commit and unstages changes
git reset --mixed HEAD~1

# Hard reset:
# Removes commit and deletes changes completely
git reset --hard HEAD~1


# =========================================================
# DISCARD UNWANTED LOCAL CHANGES
# =========================================================

# Discard unstaged file changes
git checkout -- app.txt

# Discard all local changes
git reset --hard


# =========================================================
# REVERT COMMIT (SAFE FOR SHARED BRANCH)
# =========================================================

# Create reverse commit safely
git revert COMMIT_ID

# Example:
# git revert a1b2c3d


# =========================================================
# VERIFY ROLLBACK
# =========================================================

# Check commit history
git log --oneline

# Check current status
git status

# View file content
cat app.txt


# =========================================================
# VIEW COMPLETE HISTORY
# =========================================================

git log --oneline --graph --all
