  vpc_id                  = aws_vpc.vm_vpc.id
  cidr_block              = "10.1.1.0/24"
  availability_zone       = "ap-south-1a"
  map_public_ip_on_launch = true

resource "aws_internet_gateway" "vm_igw" {
  vpc_id = aws_vpc.vm_vpc.id
}

resource "aws_route_table" "vm_rt" {
  vpc_id = aws_vpc.vm_vpc.id

  route {
    cidr_block = "0.0.0.0/0"
    gateway_id = aws_internet_gateway.vm_igw.id
  }
}

resource "aws_route_table_association" "vm_rta" {
  subnet_id      = aws_subnet.vm_subnet.id
  route_table_id = aws_route_table.vm_rt.id
}

resource "aws_key_pair" "vm_key" {
  key_name   = "mykey"
  public_key = file("C:/Users/YourUser/.ssh/id_rsa.pub")
}

resource "aws_security_group" "vm_sg" {
  name   = "vm-security-group"
  vpc_id = aws_vpc.vm_vpc.id

  ingress {
    from_port   = 22
    to_port     = 22
    protocol    = "tcp"
    cidr_blocks = ["0.0.0.0/0"]
  }

  ingress {
    from_port   = 80
    to_port     = 80
    protocol    = "tcp"
    cidr_blocks = ["0.0.0.0/0"]
  }

  egress {
    from_port   = 0
    to_port     = 0
    protocol    = "-1"
    cidr_blocks = ["0.0.0.0/0"]
  }
}

resource "aws_instance" "vm_instance" {
  ami                         = "ami-03f4878755434977f"
  instance_type               = "t3.micro"
  subnet_id                   = aws_subnet.vm_subnet.id
  vpc_security_group_ids      = [aws_security_group.vm_sg.id]
  key_name                    = aws_key_pair.vm_key.key_name
  associate_public_ip_address = true

  user_data = <<EOF
#!/bin/bash
apt update -y
apt install -y apache2
systemctl start apache2
systemctl enable apache2
echo "Hello from Terraform EC2!" > /var/www/html/index.html
EOF

  tags = {
    Name = "Terraform-EC2"
  }
}

output "vm_public_ip" {
  value = aws_instance.vm_instance.public_ip
}

  type = string

  name = "my-db-subnet-group"

  subnet_ids = [
    aws_subnet.vm_subnet.id
  ]

resource "aws_security_group" "rds_sg" {
  name   = "rds-security-group"
  vpc_id = aws_vpc.vm_vpc.id

  ingress {
    from_port       = 3306
    to_port         = 3306
    protocol        = "tcp"
    security_groups = [aws_security_group.vm_sg.id]

  egress {
    from_port   = 0
    to_port     = 0
    protocol    = "-1"
    cidr_blocks = ["0.0.0.0/0"]
  }
}

  identifier             = "terraform-mysql-db"
  allocated_storage      = 20

  engine_version         = "8.0"

  publicly_accessible    = false
  db_subnet_group_name   = aws_db_subnet_group.db_subnet_group.name
  vpc_security_group_ids = [aws_security_group.rds_sg.id]

# Recipe 20: Load Balancer

resource "aws_lb_target_group" "alb_tg" {
  name     = "alb-tg"
  port     = 80
  protocol = "HTTP"
  vpc_id   = aws_vpc.vm_vpc.id
}

resource "aws_lb" "alb" {
  name               = "terraform-alb"
  load_balancer_type = "application"

  subnets = [
    aws_subnet.vm_subnet.id
  ]

  security_groups = [aws_security_group.vm_sg.id]
}

resource "aws_lb_listener" "alb_listener" {
  load_balancer_arn = aws_lb.alb.arn
  port              = 80
  protocol          = "HTTP"

  default_action {
    type             = "forward"
    target_group_arn = aws_lb_target_group.alb_tg.arn
  }
}

output "alb_dns" {
  value = aws_lb.alb.dns_name
}

# ================================
# Recipe 23: Auto Scaling
# ================================

resource "aws_autoscaling_group" "asg" {
  desired_capacity = 2
  max_size         = 4
  min_size         = 1

  vpc_zone_identifier = [
    aws_subnet.vm_subnet.id
  ]

  launch_template {
    id      = aws_launch_template.zdd_lt.id
    version = "$Latest"
  }

  tag {
    key                 = "Name"
    value               = "asg-instance"
    propagate_at_launch = true
  }
}

output "asg_name" {
  value = aws_autoscaling_group.asg.name
}
